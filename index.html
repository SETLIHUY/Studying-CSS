១ margin គឺអ្វី?

margin គឺជាគម្លាតទទេនៅ ខាងក្រៅ ជុំវិញ element មួយ។ វាច្រាន element ផ្សេងទៀតឱ្យនៅឆ្ងាយពី element នេះ។ margin គ្មានពណ៌ខ្លួនឯងទេ។ ក្នុងរូបខាងលើ យើងដាក់ពណ៌ទឹកក្រូចដើម្បីឱ្យមើលឃើញ។

ប្រអប់នីមួយៗនៅក្នុង HTML មាន ៤ ស្រទាប់ ពីក្នុងទៅក្រៅ៖

ស្រទាប់អត្ថន័យ
contentអត្ថបទ ឬរូបភាពនៅខាងក្នុង
paddingគម្លាតខាងក្នុង រវាង content និង border
borderស៊ុមព័ទ្ធជុំវិញ
marginគម្លាតខាងក្រៅ រវាង element នេះ និង element ផ្សេង

ចងចាំ៖ padding នៅក្នុងស៊ុម ហើយមានពណ៌ផ្ទៃរបស់ element។ margin នៅក្រៅស៊ុម ហើយតែងតែថ្លា។

២ margin ទាំង ៤ ទិស

យើងអាចកំណត់ margin សម្រាប់ទិសនីមួយៗដោយឡែកពីគ្នា៖

លក្ខណៈទិសឧទាហរណ៍
margin-topខាងលើmargin-top: 20px;
margin-rightខាងស្តាំmargin-right: 10px;
margin-bottomខាងក្រោមmargin-bottom: 20px;
margin-leftខាងឆ្វេងmargin-left: 10px;
marginទាំង ៤ ទិសក្នុងមួយបន្ទាត់margin: 20px;

៣ សរសេរខ្លី (shorthand)

លក្ខណៈ margin មួយអាចទទួលតម្លៃ ១ ដល់ ៤។ ចំនួនតម្លៃកំណត់ថាទិសណាទទួលតម្លៃអ្វី៖

ការសរសេរអត្ថន័យ
margin: 20px;ទាំង ៤ ទិស ស្មើ 20px
margin: 10px 30px;លើ និងក្រោម 10px, ឆ្វេង និងស្តាំ 30px
margin: 10px 20px 30px;លើ 10px, ឆ្វេង និងស្តាំ 20px, ក្រោម 30px
margin: 10px 20px 30px 40px;លើ 10px, ស្តាំ 20px, ក្រោម 30px, ឆ្វេង 40px

ល្បិចចងចាំ៖ តម្លៃ ៤ ទៅតាមទិសទ្រនិចនាឡិកា ចាប់ពីខាងលើ៖ លើ → ស្តាំ → ក្រោម → ឆ្វេង (Top, Right, Bottom, Left)។

margin-inline និង margin-block

មានវិធីសរសេរខ្លីទៀតតាមទិសនៃអត្ថបទ៖ margin-inline ជំនួស ឆ្វេង និងស្តាំ ហើយ margin-block ជំនួស លើ និងក្រោម។

.box {
  margin-inline: auto;
  margin-block: 24px;
}

៤ margin: auto

តម្លៃ auto ឱ្យ browser ចែកទំហំទទេដែលនៅសល់ស្មើគ្នា។ ប្រសិនបើដាក់ auto ទាំងឆ្វេង និងស្តាំ element នឹងនៅចំកណ្តាល។ ត្រូវកំណត់ width (ឬ max-width) ហើយ element ត្រូវជា block។

.container {
  width: 200px;
  margin: 0 auto;
}

ចំកណ្តាលដោយ margin: 0 auto

margin: 0 auto

ច្រានទៅម្ខាងក្នុង flex

ក្នុង display: flex ការដាក់ margin-left: auto លើ item មួយ នឹងច្រាន item នោះទៅខាងស្តាំបំផុត។ វិធីនេះប្រើញឹកញាប់ក្នុង navbar។

៥ margin អវិជ្ជមាន

តម្លៃអវិជ្ជមាន (ដូចជា -18px) ទាញ element ឱ្យទៅរកទិសនោះ ហើយអាចត្រួតលើ element ផ្សេង។ ប្រើតែពេលចាំបាច់ ព្រោះពិបាកថែទាំ។

៦ margin បញ្ចូលគ្នា (collapse)

ពេល margin-bottom របស់ element មួយជួប margin-top របស់ element បន្ទាប់ តាមបញ្ឈរ ពួកវា មិនបូកគ្នាទេ។ browser យកតម្លៃធំជាងគេ។ ឧទាហរណ៍ 40px និង 30px បានជាចន្លោះ 40px មិនមែន 70px។

  • margin ខាងឆ្វេង និងស្តាំ មិនបញ្ចូលគ្នាទេ។ វាបូកគ្នាធម្មតា។

៩ លក្ខណៈផ្សេងទៀតដែលប្រើជាមួយ margin

margin ធ្វើការល្អបំផុតពេលប្រើជាមួយលក្ខណៈទាំងនេះ៖

លក្ខណៈមុខងារឧទាហរណ៍

borderស៊ុមជុំវិញ elementborder: 2px solid #333;
width / max-widthទទឹង និងទទឹងអតិបរមា (ចាំបាច់សម្រាប់ margin: auto)max-width: 600px;
height / min-heightកម្ពស់ និងកម្ពស់អប្បបរមាmin-height: 200px;

ចំណងជើង card

ករណី margin មិនដំណើរការ

element ប្រភេទ inline ដូចជា <span> និង <a> មិនទទួល margin-top និង margin-bottom ទេ។ ដើម្បីដោះស្រាយ ដាក់ display: inline-block ឬ display: block។

span.tag {
  display: inline-block;

}

១០ លំហាត់

ព្យាយាមឆ្លើយមុន ហើយចុចដើម្បីមើលចម្លើយ។

១. margin: 5px 10px 15px 20px; តើ margin-left ស្មើប៉ុន្មាន?

20px។ តម្លៃទី ៤ ជាខាងឆ្វេង (លើ → ស្តាំ → ក្រោម → ឆ្វេង)។

២. ចង់ដាក់ div ទទឹង 300px នៅចំកណ្តាល parent តើសរសេរយ៉ាងដូចម្តេច? 
div {
  width: 300px;
  margin: 0 auto;
}
៣. box A មាន margin-bottom: 25px ហើយ box B មាន margin-top: 15px។ តើចន្លោះពិតប៉ុន្មាន?

25px។ margin បញ្ចូលគ្នា ហើយយកតម្លៃធំជាង។

៤. ហេតុអ្វី margin-top លើ <span> មិនដំណើរការ?

ព្រោះ span ជា inline element។ ដាក់ display: inline-block ដើម្បីឱ្យ margin បញ្ឈរដំណើរការ។

៥. តើ margin ខុសពី padding យ៉ាងដូចម្តេច?

margin នៅខាងក្រៅ border ហើយថ្លា។ padding នៅខាងក្នុង border ហើយមានពណ៌ផ្ទៃរបស់ element។

សង្ខេប

  • margin គឺគម្លាតខាងក្រៅ ហើយមាន ៤ ទិស៖ top, right, bottom, left។
  • សរសេរខ្លីតាមទិសទ្រនិចនាឡិកា៖ លើ → ស្តាំ → ក្រោម → ឆ្វេង។
  • margin: 0 auto ធ្វើឱ្យ block ដែលមាន width នៅចំកណ្តាល។
  • margin បញ្ឈរបញ្ចូលគ្នា (យកតម្លៃធំជាង) ហើយ margin ផ្តេកបូកគ្នា។
  • ប្រើ padding, gap និង box-sizing ជាមួយ margin ដើម្បីរៀបចំ layout ឱ្យស្អាត។
